\begin{enumerate}
\item Throughout this problem we regard $[n]^d = \{1,2,\dots,n\}^d$ as a subset of $\reals^d$ in the standard way.

A \emph{geometric line} in $[n]^d$ is a set $L\subseteq[n]^d$ of $n$ distinct points which are collinear in $\reals^d$; that is, a set $L$ with $|L| = n$ for which there exist $a\in\reals^d$ and $v\in\reals^d\setminus\{0\}$ such that
\[
L \subseteq \{a + sv \mid s\in\reals\}.
\]
Note that a geometric line is a set of points, so that a line and its reversal are the same line, and note that we require $|L| = n$ exactly. We write $\calL_{n,d}$ for the $n$-uniform hypergraph on the vertex set $[n]^d$ whose edges are the geometric lines of $[n]^d$. In class we studied the case $n=8$, $d=3$, since $\calL_{8,3}$ is the hypergraph underlying $8\times8\times8$ tic-tac-toe.
\begin{enumerate}
\item How many geometric lines does $[n]^d$ have? \emph{Suggestion:} begin by characterizing exactly which subsets of $[n]^d$ are geometric lines.
\item Suppose we set each vertex of $[n]^d$ to be an X with probability $p$ and an O with probability $1-p$, independently. Say a geometric line is \emph{all-X} if every one of its $n$ vertices is set to X.

Taking $d = \alpha n$ for a constant $\alpha>0$, determine a threshold $f_\alpha(n)$ --- make it as simple as you can --- such that
\begin{itemize}
\item if $p = o(f_\alpha(n))$, then w.h.p.\ there is no all-X geometric line, while
\item if $p = \omega(f_\alpha(n))$, then w.h.p.\ there is at least one all-X geometric line.
\end{itemize}
\end{enumerate}
\end{enumerate}

\solution
\paragraph{(a)}
There are $((n+2)^d -n^d)/2$ geometric lines in $[n]^d$.

All geometric lines are either increasing, decreasing or constant in each dimension, and there are $(n+2)^d$ such choices.
However, geometric lines cannot be constant in all dimensions, so we need to subtract $n^d$ such choices ($n$ constant values for each dimension).
Furthermore, each line has two endpoints we can start selecting points from, so it has two representations as a choice of increasing/decreasing/constant coordinates in each dimension, which is why we divide the final expression by 2.

\paragraph{(b)}

\ul{$p$ sufficiently small:}
\begin{align*}
    \Pr(\text{there exists an all-}X \text{ line}) &\leq \sum_{\ell} \Pr(\ell \text{ is all-}X)&& (\text{Union Bound}) \\
    & = \frac{1}{2}((n+2)^{\alpha n}-n^{\alpha n}) p^{ n} \\
    & \leq (p(n+2)^{\alpha})^n.
\end{align*}

Thus, if $p= o\left(\frac{1}{(n+2)^\alpha}\right)$, the above term goes to zero as $n\to \infty$.
Hence, w.h.p there is are no all-$X$ lines.

\ul{$p$ sufficiently big:}
We construct a large set of lines $L$ which do now share vertices. 
Hence, the event that each line in $L$ is all-$X$ is independent.

Let $L$ be the set of lines which are constant in the first $d-1$ dimensions and increasing in the $d$th dimension. 
Then there are $n^{d-1} = n^{\alpha n-1}$ such lines.

The probability that all lines are not all-$X$ is upper bounded by the probability that all lines in $L$ are not all-$X$.

\begin{align*}
    \Pr(\text{all lines in }L\text{ are not all-}X) &= \prod_{\ell\in L}\Pr (\ell\text{ is not all-}X) \\
    & =(1-p^n)^{n^{\alpha n - 1}}
\end{align*}

Using the Taylor's series approximation $\ln (1-x)\approx -x \Rightarrow 1-x \approx \exp(-x)$ for $x=p^n$.

\begin{align*}
    \Pr(\text{all lines in }L\text{ are not all-}X) &\approx \exp\left(-p^n{n^{\alpha n - 1}}\right) \\
    & = \exp\left(-\frac{1}{n}(pn^\alpha)^n\right).
\end{align*}

Thus, if $p = \omega\left(\frac{1}{n^\alpha}\right)$, as $n\to \infty$, $\frac{1}{n}(pn^\alpha)^n \to \infty$, so the above term tends to $0$. 
Hence, w.h.p. there is at least one all-$X$ line. 

\begin{enumerate}[resume]
\item Let $k\geq 3$ be a constant, and suppose $np\to\infty$. Prove that
\[
\Pr\bigl(G(n,p)\text{ contains a $k$-cycle}\bigr)\to 1 \quad\text{as } n\to\infty.
\]

\emph{Note:} In lecture we proved a general threshold result for the appearance of an arbitrary fixed subgraph $H$ in $G(n,p)$. Do not simply quote that theorem. Give a self-contained argument for this special case: let $X$ be the number of $k$-cycles in $G(n,p)$, compute $\EE(X)$ and bound $\Var(X)$ directly, and apply the second moment method to these computations.
\end{enumerate}

\solution
Let $X$ be the number of $k$-cycles in $G(n,p)$. 
We will first compute $\EE[X]$ and $\Var(X)$, then use Chebyshev's inequality.
We can express the expectation of $X$ as
\begin{align*}
    \EE[X] &=\sum_{k\text{-cycle }C} \Pr(C\in G(n,p)) && (\text{linearity of expectation}) \\
    &=\underbrace{\binom{n}{k}\cdot\frac{k!}{2k}}_{\text{number of }k\text{-cycles}}\cdot p^k \\
    &\geq c_k(np)^{2k}.
\end{align*}
for some constant $c_k$ and where we use the inequality $\binom{n}{k}\geq c'_k n^k$ from some constant $c'_k$. 
For each $k$-cycle $C$ in $K_n$, let $\xi_C$ be the indicator random variable on the event that $C\in G(n,p)$. 
It follows that $X=\sum_{C}\xi_C$ and therefore we can write its square as

\begin{align*}
    \EE[X]^2 &= \left(\sum_{C}\EE[\xi_C]\right)^2   && \text{(linearity of expectation)}\\
    &=\sum_{\substack{k\text{-cycles} \\ C_1,C_2}}\EE[\xi_{C_1}]\EE[\xi_{C_2}].
\end{align*}

Next, we express $\EE[X^2]$ as
\begin{align*}
    \EE[X^2]  &= \sum_{\substack{k\text{-cycles} \\ C_1,C_2}} \EE[\xi_{C_1}\xi_{C_2}] \\
    &= \sum_{h=0}^k\sum_{\substack{k\text{-cycles } C_1,C_2 \\ \text{with overlap of }h \text{ vertices}}} \EE[\xi_{C_1}\xi_{C_2}] \\
    &\qquad\qquad\qquad\qquad\qquad (\text{split by number of overlapping vertices } h) \\
    &= \underbrace{\sum_{\substack{k\text{-cycle } C}} \EE[\xi_{C}\xi_{C}] + \sum_{\substack{k\text{-cycles } C_1,C_2 \\ \text{with no overlapping vertices}}} \EE[\xi_{C_1}\xi_{C_2}]}_{(1)} + \underbrace{\sum_{h=1}^{k-1}\sum_{\substack{k\text{-cycles } C_1,C_2 \\ \text{with overlap of }h \text{ vertices}}} \EE[\xi_{C_1}\xi_{C_2}]}_{(2)}. 
\end{align*}

Due to the independence of the events in the products of $(1)$, for each expectation we have that $\EE[\xi_{C_1}\xi_{C_2}] = \EE[\xi_{C_1}]\EE[\xi_{C_2}]$. 
Hence, $(1)\leq \EE[X]^2$.
For $(2)$, consider each $h$ term separately. We get
\begin{align*}
    \sum_{\substack{k\text{-cycles } C_1,C_2 \\ \text{with overlap of }h \text{ vertices}}} \EE[\xi_{C_1}\xi_{C_2}] &= \sum_{\substack{k\text{-cycles } C_1,C_2 \\ \text{with overlap of }h \text{ vertices}}} \EE[\xi_{C_1}\xi_{C_2}] \\
    &=\sum_{\substack{k\text{-cycles } C_1,C_2 \\ \text{with overlap of }h \text{ vertices}}} \Pr(C_1,C_2\in G(n,p)) 
\end{align*}
There are at most $\binom{n}{2k-h}\binom{2k-h}{k}\leq c^{(2)}_k n^{2k-h}$ terms in the sum for some constant $c^{(2)}_k$. 
For each such $C_1,C_2$, $C_1$ and $C_2$ combined have $|C_1\cup C_2|\geq 2k-(h-1) = 2k-h+1$ edges. 
Hence, we can upper bound the probability that they are contained in $G(n,p)$ as

\begin{align*}
    \Pr(C_1,C_2\in G(n,p)) = p^{|C_1\cup C_2|} \leq p^{2k-h+1}.
\end{align*}
Thus, we can bound (2) by
\begin{align*}
    \sum_{\substack{k\text{-cycles } C_1,C_2 \\ \text{with overlap of }h \text{ vertices}}} \Pr(C_1,C_2\in G(n,p))  & \leq c^{(2)}_k n^{2k-h} p^{2k-h+1} \\
    & = c^{(2)}_k p (np)^{2k-h}.
\end{align*}
Finally, by Chebyshev's inequality we have that
\begin{align*}
    \Pr(X=0)\leq \Pr(|X-\EE[X]|\geq \EE[X]) &\leq \frac{\Var(X)}{\EE[X]^2} \\
    &= \frac{\EE[X^2] -\EE[X]^2}{\EE[X]^2} \\
    &\leq \frac{\sum_{h=1}^{2k-1} c^{(2)}_k p (np)^{2k-h}}{c_k(np)^{2k}}.
\end{align*}
All terms in the sum are $o((np)^{2k})$, so the above term goes to zero as $n\to\infty$. 
Therefore, the probability that $G(n,p)$ contains a $k$-cycle goes to one as $n\to\infty$.

\begin{align*}

We will define a random variable that is the discrepency (the difference in $X$'s and $O$'s) for each line. 
We will upper bound the probability that the discrepency for each line is higher than the above bound using Chernoff. 
Finally, we will bound the probability that any line has high discrepency using the union bound.

Select whether each coordinate is colored $X$ or $O$ independently with probability $\frac{1}{2}$.
For each geometric line $\ell$, define the following indicator random variable
\begin{align*}
    \xi_{\ell,j} = \begin{cases}
        1 &\text{if the }j\text{th point in the line is assigned $X$}, \\
        -1 &\text{otherwise,}
    \end{cases}
\end{align*}
for each $j\in [n]$. For all lines $\ell$ and all $j\in[n]$, we have that $\EE[\xi_{\ell,j}]=0$ and $\lvert\xi_{\ell,j}\rvert \leq 1$. 
Define the random variable $X_\ell=\sum_{j\in[n]}\xi_{\ell,j}$. By the answer to Question 3, we get that 
\begin{align*}
    \Pr(|X_\ell|\geq \sqrt{10nd\log n})&\leq 2e^{-\frac{10nd\log n}{2n}} \\
    & = 2e^{-5d\log n} \\
    & = 2n^{-5d}.
\end{align*}

\begin{align*}
    \Pr(\text{any line has }\lvert X_\ell\rvert\geq \sqrt{10nd\log n}) &\leq \sum_{\text{line }\ell}\Pr(|X_\ell|\geq \sqrt{10nd\log n}) && \text{(union bound)} \\
    &\leq \frac{1}{2}\left((n+2)^d-n^d\right)\cdot 2n^{-5d} \\
    &\leq n^d\cdot n^{-5d} \\
    &= n^{-4d},
\end{align*}
where the second inequality uses our count of lines from Question 1.

Hence, for sufficiently large $n$, the probability that any line has $\lvert X_\ell\rvert\geq \sqrt{10nd\log n}$ is lower than $1$, so there must exist some coloring such that all lines have $\lvert X_\ell\rvert < \sqrt{10nd\log n}$.

\begin{align*}

The function $f$ is 1-coordinate Lipschitz, since each additional point can only increase the number of close pairs by two, i.e., the pairs involving the point immediately before and after it in the sorted order.
Let the random variable of the value of $f$ be $X:= f(Y_1,.., Y_d)$.

\end{lemma}

    By symmetry, the probability that each coordinate is small is equal to the probability that the first coordinate is small.
    Calculating this probability:
    \begin{align*}
        \Pr(\text{first coordinate is small}) &= 1- \Pr\left(\text{all points }\geq \frac{1}{d+1}\right) \\
        &= 1- \left(\frac{d}{d+1}\right)^d \\
        &= 1- \left(1-\frac{1}{d+1}\right)^d \\
        &\leq 1-\frac{1}{e}\approx 0.632.
    \end{align*}
